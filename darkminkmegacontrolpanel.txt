    Header.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPosition = Window.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            Window.Position = UDim2.new(startPosition.X.Scale, startPosition.X.Offset + delta.X, startPosition.Y.Scale, startPosition.Y.Offset + delta.Y)
        end
    end)
end

local function getCharacter()

    Character = Player.Character

    if not Character then return end



    Humanoid = Character:FindFirstChildOfClass("Humanoid")

    Root = Character:FindFirstChild("HumanoidRootPart")

end


local function restoreNoClip()
    for part, originalValue in pairs(OriginalCanCollide) do
        if part and part.Parent then
            part.CanCollide = originalValue
        end
    end
    table.clear(OriginalCanCollide)
end

local function applyNoClipPart(part)
    if NoClipEnabled and part and part:IsA("BasePart") then
        if OriginalCanCollide[part] == nil then
            OriginalCanCollide[part] = part.CanCollide
        end
        part.CanCollide = false
    end
end

local function bindNoClipCharacter(character)
    if NoClipDescendantConnection then
        NoClipDescendantConnection:Disconnect()
        NoClipDescendantConnection = nil
    end
    restoreNoClip()
    NoClipCharacter = character
    if not character or not NoClipEnabled then return end

    for _, descendant in ipairs(character:GetDescendants()) do
        applyNoClipPart(descendant)
    end
    NoClipDescendantConnection = character.DescendantAdded:Connect(applyNoClipPart)
end

local function setNoClip(enabled)
    NoClipEnabled = enabled
    getCharacter()
    if enabled then
        bindNoClipCharacter(Character)
        -- Apply immediately after enabling so every current body part is updated.
        if Character then
            for _, descendant in ipairs(Character:GetDescendants()) do
                applyNoClipPart(descendant)
            end
        end
    else
        if NoClipDescendantConnection then
            NoClipDescendantConnection:Disconnect()
            NoClipDescendantConnection = nil
        end
        NoClipCharacter = nil
        restoreNoClip()
    end
end

-- Reassert collision state continuously. Some character/game scripts reset
-- CanCollide after teleports or when assembling/replacing character parts.
local function refreshNoClip()
    if not NoClipEnabled then return end

    getCharacter()
    if not Character then return end

    if NoClipCharacter ~= Character then
        bindNoClipCharacter(Character)
    end

    for _, descendant in ipairs(Character:GetDescendants()) do
        if descendant:IsA("BasePart") and descendant.CanCollide then
            applyNoClipPart(descendant)
        elseif descendant:IsA("BasePart") and OriginalCanCollide[descendant] == nil then
            -- Remember the original value even when it was already non-collidable.
            OriginalCanCollide[descendant] = false
        end
    end
end


local function restoreInvisible()

    for object, value in pairs(OriginalLTM) do

        if object and object.Parent then

            object.LocalTransparencyModifier = value

        end

    end

    table.clear(OriginalLTM)

end



local function setInvisible(enabled)

    InvisibleEnabled = enabled

    getCharacter()

    if not Character then return end



    if enabled then

        table.clear(OriginalLTM)



        for _, object in ipairs(Character:GetDescendants()) do

            if object:IsA("BasePart") then

                OriginalLTM[object] = object.LocalTransparencyModifier

                object.LocalTransparencyModifier = 1

            end

        end

    else

        restoreInvisible()

    end

end



local function applySpeed()

    if Humanoid and SpeedEnabled then

        Humanoid.WalkSpeed = SpeedValue

    end

end



local function applyJump()

    if not Humanoid or not JumpEnabled then return end



    if Humanoid.UseJumpPower then

        Humanoid.JumpPower = JumpValue

    else

        Humanoid.JumpHeight = math.clamp(JumpValue / 2, 1, 250)

    end

end


-- FLY: camera-relative movement at the selected studs/second speed.
local function stopFly()
    FlyEnabled = false

    if FlyConnection then
        FlyConnection:Disconnect()
        FlyConnection = nil
    end

    if FlyVelocity then
        FlyVelocity:Destroy()
        FlyVelocity = nil
    end

    if FlyGyro then
        FlyGyro:Destroy()
        FlyGyro = nil
    end

    getCharacter()
    if Humanoid then
        Humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
    end
end

local function startFly()
    getCharacter()
    if not Root or not Humanoid or Humanoid.Health <= 0 then
        return false
    end

    stopFly()
    getCharacter()
    if not Root or not Humanoid then
        return false
    end

    FlyEnabled = true

    FlyVelocity = Instance.new("BodyVelocity")
    FlyVelocity.Name = "CharacterFlyVelocity"
    FlyVelocity.MaxForce = Vector3.new(1e9, 1e9, 1e9)
    FlyVelocity.P = 1e5
    FlyVelocity.Velocity = Vector3.zero
    FlyVelocity.Parent = Root

    FlyGyro = Instance.new("BodyGyro")
    FlyGyro.Name = "CharacterFlyGyro"
    FlyGyro.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
    FlyGyro.P = 1e5
    FlyGyro.CFrame = Root.CFrame
    FlyGyro.Parent = Root

    Humanoid:ChangeState(Enum.HumanoidStateType.Physics)

    FlyConnection = RunService.RenderStepped:Connect(function()
        if not FlyEnabled or not Root or not Root.Parent
            or not Humanoid or Humanoid.Health <= 0 then
            stopFly()
            return
        end

        local camera = workspace.CurrentCamera
        if not camera then return end

        local move = Vector3.zero
        local forward = camera.CFrame.LookVector
        local right = camera.CFrame.RightVector

        if UIS:IsKeyDown(Enum.KeyCode.W) then move += forward end
        if UIS:IsKeyDown(Enum.KeyCode.S) then move -= forward end
        if UIS:IsKeyDown(Enum.KeyCode.D) then move += right end
        if UIS:IsKeyDown(Enum.KeyCode.A) then move -= right end
        if UIS:IsKeyDown(Enum.KeyCode.Space) then move += Vector3.yAxis end
        if UIS:IsKeyDown(Enum.KeyCode.LeftControl)
            or UIS:IsKeyDown(Enum.KeyCode.RightControl) then
            move -= Vector3.yAxis
        end

        if move.Magnitude > 0 then
            move = move.Unit
        end

        FlyVelocity.Velocity = move * FlySpeedValue
        FlyGyro.CFrame = camera.CFrame
    end)

    return true
end

local function startAuto()



    if #locations == 0 then

        status.Text = "STATUS: NO LOCATIONS"

        return

    end



    if autoTP then return end



    autoTP = true

    autoToken += 1



    local myToken = autoToken



    autoButton.Text = "AUTO TP: ON"

    autoButton.BackgroundColor3 = Color3.fromRGB(20,160,85)



    status.Text = "STATUS: STARTING..."



    task.spawn(function()



        while autoTP and autoToken == myToken do



            local total = #locations



            if total == 0 then

                stopAuto()

                break

            end



            for i = 1,total do



                if not autoTP or autoToken ~= myToken then

                    break

                end



                local loc = locations[i]



                if loc then



                    local character = player.Character

                    local humanoid = character and character:FindFirstChildOfClass("Humanoid")



                    if character and humanoid and humanoid.Health > 0 then



                        -- TELEPORT TO LOCATION

                        character:PivotTo(

                            CFrame.new(

                                loc.position + Vector3.new(0,3,0)

                            )

                        )



                        status.Text =

                            "STATUS: LOCATION "

                            ..i.." / "..total



                    else



                        status.Text = "STATUS: CHARACTER NOT READY"



                    end



                end



                task.wait(tpDelay)



            end



            if not repeatTP then



                stopAuto()

                break



            end



            task.wait(0.01)



        end



    end)



end

--==================================================

-- SAVE / CHANGE / RESTORE

--==================================================



local function Save(obj, property)

    if not Saved[obj] then

        Saved[obj] = {}

    end



    if Saved[obj][property] ~= nil then

        return

    end



    local success, value = pcall(function()

        return obj[property]

    end)



    if success then

        Saved[obj][property] = {value = value}

    end

end



local function Change(obj, property, value)

    Save(obj, property)



    pcall(function()

        obj[property] = value

    end)

end



local function Restore()

    for obj, properties in pairs(Saved) do

        if obj then

            for property, stored in pairs(properties) do

                pcall(function()

                    obj[property] = stored.value

                end)

            end

        end

    end



    Saved = {}



    if OriginalZoom ~= nil then

        pcall(function()

            Player.CameraMaxZoomDistance = OriginalZoom

        end)

        OriginalZoom = nil

    end

end



--==================================================

-- CHARACTER DETECTION

--==================================================



local function GetPlayerCharacterFromObject(obj)

    for _, targetPlayer in ipairs(Players:GetPlayers()) do

        local character = targetPlayer.Character

        if character and (obj == character or obj:IsDescendantOf(character)) then

            return character

        end

    end



    return nil

end



local function IsInsideAccessory(obj, character)

    local current = obj



    while current and current ~= character do

        if current:IsA("Accessory") then

            return true

        end

        current = current.Parent

    end



    return false

end



--==================================================

-- DISABLE EFFECTS

--==================================================



local function DisableEffect(obj)

    if obj:IsA("PostEffect")

        or obj:IsA("ParticleEmitter")

        or obj:IsA("Trail")

        or obj:IsA("Beam")

        or obj:IsA("Fire")

        or obj:IsA("Smoke")

        or obj:IsA("Sparkles")

        or obj:IsA("Highlight")

        or obj:IsA("PointLight")

        or obj:IsA("SpotLight")

        or obj:IsA("SurfaceLight")

        or obj:IsA("Clouds") then



        Change(obj, "Enabled", false)

    end

end



--==================================================

-- HIDE CHARACTER APPEARANCE (LOCAL VISUAL ONLY)

--==================================================



local function HideCharacterAppearance(obj, character)

    if obj:IsA("BasePart") and IsInsideAccessory(obj, character) then

        Change(obj, "LocalTransparencyModifier", 1)

    end



    if obj:IsA("Shirt") then

        Change(obj, "ShirtTemplate", "")

    elseif obj:IsA("Pants") then

        Change(obj, "PantsTemplate", "")

    elseif obj:IsA("ShirtGraphic") then

        Change(obj, "Graphic", "")

    end



    if (obj:IsA("Decal") or obj:IsA("Texture"))

        and obj:IsDescendantOf(character) then

        Change(obj, "Transparency", 1)

    end



    if obj:IsA("BodyColors") then

        for _, property in ipairs({

            "HeadColor3",

            "TorsoColor3",

            "LeftArmColor3",

            "RightArmColor3",

            "LeftLegColor3",

            "RightLegColor3"

        }) do

            Change(obj, property, GREY)

        end

    end



    if obj:IsA("SurfaceAppearance") then

        Change(obj, "Color", GREY)

        Change(obj, "ColorMap", "")

        Change(obj, "MetalnessMap", "")

        Change(obj, "NormalMap", "")

        Change(obj, "RoughnessMap", "")

    end

end



--==================================================

-- FLATTEN WORLD VISUALS WITHOUT DELETING BLOCKS

--==================================================



local function FlattenBlock(obj)

    if obj:IsA("BasePart") then

        Change(obj, "Color", GREY)

        Change(obj, "Material", Enum.Material.SmoothPlastic)

        Change(obj, "Reflectance", 0)

        Change(obj, "CastShadow", false)



        if obj:IsA("MeshPart") then

            Change(obj, "RenderFidelity", Enum.RenderFidelity.Performance)

        end



    elseif obj:IsA("Decal") or obj:IsA("Texture") then

        Change(obj, "Transparency", 1)



    elseif obj:IsA("SurfaceAppearance") then

        Change(obj, "Color", GREY)

        Change(obj, "ColorMap", "")

        Change(obj, "MetalnessMap", "")

        Change(obj, "NormalMap", "")

        Change(obj, "RoughnessMap", "")

    end

end



--==================================================

-- APPLY OPTIMIZATION TO AN OBJECT

--==================================================



local function OptimizeObject(obj)

    if not State.FPS or not obj then

        return

    end



    DisableEffect(obj)

    FlattenBlock(obj)



    local character = GetPlayerCharacterFromObject(obj)

    if character then

        HideCharacterAppearance(obj, character)

    end

end



local function OptimizeCharacter(character)

    if not State.FPS or not character then

        return

    end



    for _, obj in ipairs(character:GetDescendants()) do

        DisableEffect(obj)

        FlattenBlock(obj)

        HideCharacterAppearance(obj, character)

    end

end



--==================================================

-- APPLY FPS BOOST

--==================================================



local function ApplyFPS()

    Change(Lighting, "GlobalShadows", false)



    pcall(function()

        Change(Workspace.Terrain, "Decoration", false)

    end)



    for _, obj in ipairs(Lighting:GetDescendants()) do

        OptimizeObject(obj)

    end



    for _, obj in ipairs(Workspace:GetDescendants()) do

        OptimizeObject(obj)

    end



    for _, targetPlayer in ipairs(Players:GetPlayers()) do

        if targetPlayer.Character then

            OptimizeCharacter(targetPlayer.Character)

        end

    end



    if OriginalZoom == nil then

        OriginalZoom = Player.CameraMaxZoomDistance

    end



    pcall(function()

        Player.CameraMaxZoomDistance = math.min(OriginalZoom * 10, 10000)

    end)

end

-- GET PUBLIC SERVERS

--====================================================



local function getServers()

    local cursor = ""

    local servers = {}



    for page = 1, 5 do

        local url =

            "https://games.roblox.com/v1/games/"

            .. PlaceId

            .. "/servers/Public?sortOrder=Asc&limit=100"



        if cursor ~= "" then

            url = url .. "&cursor=" .. HttpService:UrlEncode(cursor)

        end



        local response



        local success, result = pcall(function()

            return game:HttpGet(url)

        end)



        if not success then

            error("Không thể truy cập danh sách server")

        end



        response = HttpService:JSONDecode(result)



        for _, server in ipairs(response.data or {}) do

            if server.id

                and server.id ~= CurrentJobId

                and server.playing < server.maxPlayers then



                table.insert(servers, server)

            end

        end



        cursor = response.nextPageCursor or ""



        if cursor == "" then

            break

        end

    end



    return servers

end

--==================================================

-- MULTI SELECT SYSTEM

--==================================================



local function getSelectedIndex(plr)



    for i,player in ipairs(SelectedPlayers) do



        if player == plr then

            return i

        end



    end



    return nil

end



local function updateTarget()



    if #SelectedPlayers == 0 then



        Target = nil

        CurrentIndex = 1



        return



    end



    CurrentIndex = math.clamp(

        CurrentIndex,1,#SelectedPlayers

    )



    Target = SelectedPlayers[CurrentIndex]



end



local function toggleSelection(plr)



    local index = getSelectedIndex(plr)



    if index then



        table.remove(SelectedPlayers,index)



        if index < CurrentIndex then



            CurrentIndex -= 1



        elseif index == CurrentIndex then



            CurrentIndex = math.min(

                CurrentIndex,#SelectedPlayers

            )



        end



    else



        table.insert(SelectedPlayers,plr)



    end



    updateTarget()



end



--==================================================

-- ESP CLEANUP

--==================================================



local function removeESP(plr)



    local data = ESPObjects[plr]



    if data then



        if data.highlight then

            data.highlight:Destroy()

        end



        if data.billboard then

            data.billboard:Destroy()

        end



        ESPObjects[plr] = nil



    end



end



--==================================================

-- ESP CREATE

--==================================================



local function createESP(plr)



    removeESP(plr)



    if not ESPEnabled then return end

    if not ESPTargets[plr] then return end



    local character = plr.Character



    if not character then return end



    local head = character:FindFirstChild("Head")

    local root = character:FindFirstChild("HumanoidRootPart")



    if not head or not root then return end



    local highlight = Instance.new("Highlight")



    highlight.Name = "PlayerManagerESP"

    highlight.Adornee = character

    highlight.FillColor = Color3.fromRGB(30,170,255)

    highlight.OutlineColor = Color3.fromRGB(255,255,255)

    highlight.FillTransparency = 0.65

    highlight.OutlineTransparency = 0

    highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop

    highlight.Parent = character



    local billboard = Instance.new("BillboardGui")



    billboard.Name = "PlayerManagerDistance"

    billboard.Adornee = head

    billboard.Size = UDim2.fromOffset(190,35)

    billboard.StudsOffset = Vector3.new(0,2.5,0)

    billboard.AlwaysOnTop = true

    billboard.Parent = gui



    local label = Instance.new("TextLabel")



    label.Size = UDim2.fromScale(1,1)

    label.BackgroundTransparency = 1

    label.TextColor3 = Color3.fromRGB(80,200,255)

    label.TextStrokeTransparency = 0

    label.Font = Enum.Font.GothamBold

    label.TextSize = 14

    label.Parent = billboard



    ESPObjects[plr] = {

        highlight = highlight,

        billboard = billboard,

        label = label

    }



end



--==================================================

-- ESP DISTANCE UPDATE

--==================================================



espConnection = RunService.RenderStepped:Connect(function()



    for plr,data in pairs(ESPObjects) do



        if not ESPEnabled

            or not ESPTargets[plr]

            or plr.Parent ~= Players then



            removeESP(plr)



        else



            local myCharacter = LocalPlayer.Character

            local targetCharacter = plr.Character



            local myRoot = myCharacter

                and myCharacter:FindFirstChild("HumanoidRootPart")



            local targetRoot = targetCharacter

                and targetCharacter:FindFirstChild("HumanoidRootPart")



            if myRoot and targetRoot then



                local dist = (

                    myRoot.Position-targetRoot.Position

                ).Magnitude



                data.label.Text =

                    plr.DisplayName

                    .." | "

                    ..math.floor(dist)

                    .." STUDS"



            else



                removeESP(plr)



            end



        end



    end



end)



--==================================================

local function togglePlayerESP(plr)



    ESPTargets[plr] = not ESPTargets[plr]



    if ESPTargets[plr] then



        createESP(plr)



    else



        removeESP(plr)



    end



end

--==================================================

-- FOLLOW LOOP

--==================================================



followConnection = RunService.Heartbeat:Connect(function()



    if not Following or not Target then

        return

    end



    if Target.Parent ~= Players then



        Following = false

        updateTarget()

        refresh()



        return



    end



    local targetCharacter = Target.Character

    local myCharacter = LocalPlayer.Character



    if not targetCharacter or not myCharacter then

        return

    end



    local targetRoot =

        targetCharacter:FindFirstChild("HumanoidRootPart")



    local myRoot =

        myCharacter:FindFirstChild("HumanoidRootPart")



    if not targetRoot or not myRoot then

        return

    end



    local destination =

        targetRoot.CFrame * CFrame.new(0,0,Distance)



    myCharacter:PivotTo(

        CFrame.lookAt(

            destination.Position,

            targetRoot.Position

        )

    )



end)

--==============================================================
-- LOCK MODULE WINDOW CONTROLS
-- Only the MASTER window may minimize, maximize, or close.
-- Inner module controls are hidden and disabled; their content
-- remains controlled by the master tabs and follows the master.
--==============================================================
do
    local blockedNames = {
        minimize = true,
        minim = true,
        maximize = true,
        maxim = true,
        close = true,
        closebutton = true,
    }
    local blockedGlyphs = {
        ["—"] = true,
        ["–"] = true,
        ["-"] = true,
        ["+"] = true,
        ["□"] = true,
        ["×"] = true,
        ["X"] = true,
    }

    local modulePages = {PageCharacter, PageLocation, PageFPS, PageServer, PagePlayer}
    for _, page in ipairs(modulePages) do
        for _, object in ipairs(page:GetDescendants()) do
            if object:IsA("GuiButton") then
                local lowerName = string.lower(object.Name or "")
                local lowerText = object:IsA("TextButton") and string.lower(object.Text or "") or ""
                local isWindowControl = blockedNames[lowerName]
                    or string.find(lowerName, "minimiz", 1, true)
                    or string.find(lowerName, "maximiz", 1, true)
                    or string.find(lowerName, "close", 1, true)
                    or blockedGlyphs[object:IsA("TextButton") and object.Text or ""]
                    or blockedGlyphs[lowerText]

                if isWindowControl then
                    object.Visible = false
                    object.Active = false
                    object.AutoButtonColor = false
                end
            end
        end
    end
end
